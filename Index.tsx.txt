import { useState, useEffect, useCallback } from 'react';
import { useAuth } from '@/contexts/AuthContext';
import { Subject, Topic, Note } from '@/lib/types';
import * as store from '@/lib/store';
import Login from './Login';
import Sidebar from '@/components/Sidebar';
import NotesList from '@/components/NotesList';
import NoteEditor from '@/components/NoteEditor';
import { BookOpen, Menu } from 'lucide-react';

const Index = () => {
  const { user } = useAuth();
  const [subjects, setSubjects] = useState<Subject[]>([]);
  const [topics, setTopics] = useState<Topic[]>([]);
  const [notes, setNotes] = useState<Note[]>([]);
  const [selectedSubject, setSelectedSubject] = useState<Subject | null>(null);
  const [selectedTopic, setSelectedTopic] = useState<Topic | null>(null);
  const [selectedNote, setSelectedNote] = useState<Note | null>(null);
  const [mobileMenu, setMobileMenu] = useState(false);
  const [refreshKey, setRefreshKey] = useState(0);

  const refresh = useCallback(() => setRefreshKey(k => k + 1), []);

  useEffect(() => {
    if (!user) return;
    setSubjects(store.getSubjects(user.id));
  }, [user, refreshKey]);

  useEffect(() => {
    if (!selectedSubject) { setTopics([]); return; }
    setTopics(store.getTopics(selectedSubject.id));
  }, [selectedSubject, refreshKey]);

  useEffect(() => {
    if (!selectedTopic) { setNotes([]); return; }
    setNotes(store.getNotes(selectedTopic.id));
  }, [selectedTopic, refreshKey]);

  // Re-read selected note from store on refresh
  useEffect(() => {
    if (selectedNote) {
      const fresh = store.getNotes(selectedNote.topicId).find(n => n.id === selectedNote.id);
      if (fresh) setSelectedNote(fresh);
      else setSelectedNote(null);
    }
  }, [refreshKey]);

  const handleSelectSubject = (s: Subject) => {
    setSelectedSubject(s);
    setSelectedTopic(null);
    setSelectedNote(null);
  };

  const handleDeleteNote = () => {
    if (!selectedNote) return;
    store.deleteNote(selectedNote.id);
    setSelectedNote(null);
    refresh();
  };

  if (!user) return <Login />;

  return (
    <div className="flex h-screen bg-background overflow-hidden">
      <Sidebar
        subjects={subjects}
        selectedSubject={selectedSubject}
        selectedTopic={selectedTopic}
        topics={topics}
        onSelectSubject={handleSelectSubject}
        onSelectTopic={t => { setSelectedTopic(t); setSelectedNote(null); }}
        onRefresh={refresh}
        mobileOpen={mobileMenu}
        onMobileClose={() => setMobileMenu(false)}
      />

      {selectedTopic && selectedSubject ? (
        <div className="flex flex-1 min-w-0">
          {/* Notes list - hide on mobile when note is selected */}
          <div className={selectedNote ? 'hidden md:flex' : 'flex flex-1 md:flex-none'}>
            <NotesList
              topic={selectedTopic}
              subject={selectedSubject}
              notes={notes}
              selectedNote={selectedNote}
              onSelectNote={setSelectedNote}
              onRefresh={refresh}
              onMenuToggle={() => setMobileMenu(true)}
            />
          </div>

          {/* Editor */}
          {selectedNote ? (
            <div className="flex-1 min-w-0">
              <NoteEditor
                note={selectedNote}
                onBack={() => setSelectedNote(null)}
                onUpdate={refresh}
                onDelete={handleDeleteNote}
              />
            </div>
          ) : (
            <div className="hidden md:flex flex-1 items-center justify-center">
              <div className="text-center">
                <BookOpen className="w-12 h-12 text-muted-foreground/20 mx-auto mb-3" />
                <p className="text-muted-foreground">Select or create a note</p>
              </div>
            </div>
          )}
        </div>
      ) : (
        <div className="flex-1 flex items-center justify-center p-4">
          <div className="text-center max-w-sm">
            <button onClick={() => setMobileMenu(true)} className="md:hidden mb-4 text-muted-foreground">
              <Menu className="w-6 h-6" />
            </button>
            <div className="w-16 h-16 rounded-2xl bg-primary/10 flex items-center justify-center mx-auto mb-4">
              <BookOpen className="w-8 h-8 text-primary" />
            </div>
            <h1 className="text-2xl font-bold text-foreground mb-2">Topper Notes Lite</h1>
            <p className="text-muted-foreground text-sm">
              Create a subject in the sidebar to get started. Organize your notes by Subject → Topic → Notes.
            </p>
          </div>
        </div>
      )}
    </div>
  );
};

export default Index;
